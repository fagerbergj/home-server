# torch + torchaudio pinned to 2.3.1: pyannote.audio 3.3 imports
# torchaudio.AudioMetaData which was removed in torchaudio 2.4. Solving torch
# and pyannote in the same pip resolution pass prevents pyannote's transitive
# deps from silently upgrading torchaudio past our pin.
torch==2.3.1
torchaudio==2.3.1

# pyannote 3.4 switched to torchcodec which needs FFmpeg 6+ shared libs that
# Ubuntu 22.04 doesn't ship. 3.3.2 uses torchaudio for decoding.
pyannote.audio==3.3.2

# huggingface_hub 1.0 removed `use_auth_token` (only `token` remains), but
# pyannote 3.3.2 still calls hf_hub_download(use_auth_token=...) internally.
# Pin <1.0 so the deprecated kwarg keeps working.
huggingface_hub<1.0
