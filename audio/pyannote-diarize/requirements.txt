# torch + torchaudio + torchvision come from the ROCm base image. The
# Dockerfile generates a pip constraints file from `pip freeze` at build time
# and passes it to `pip install`, so pip leaves them at the exact ROCm-built
# versions instead of pulling CUDA-only wheels from PyPI as transitive deps.

# pyannote.audio 3.4+ uses torchcodec (FFmpeg 6 shared libs, installed via PPA
# in the Dockerfile) instead of torchaudio, so the torchaudio API removals in
# 2.5+ are no longer a problem. 3.4 also handles PyTorch 2.6+ weights_only.
pyannote.audio>=3.4.0

# huggingface_hub 1.0 removed `use_auth_token`. pyannote 3.4 still uses the
# kwarg in places, so stay on <1.0 until pyannote drops it.
huggingface_hub<1.0

fastapi
uvicorn[standard]
python-multipart

numpy

# soundfile reads our pre-converted 16 kHz mono WAV into a tensor for the
# pyannote dict input. pyannote 3.4 dropped soundfile as a direct dep when
# it switched to torchcodec, so we install it ourselves.
soundfile>=0.12.1
